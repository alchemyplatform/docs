export const LoadCustomApp = () => {
  const scriptId = 'custom-app-script'
  React.useEffect(() => {
    // Check if script is already loaded
    if (document.getElementById(scriptId)) {
      return
    }
    
    // Find base URL from any Fern asset link in DOM (scripts won't exist yet)
    const assetLink = document.querySelector('link[href*="/assets/"]')
    if (assetLink) {
      const assetSrc = assetLink.href
      const baseUrl = assetSrc.substring(0, assetSrc.lastIndexOf('/assets/'))
      const scriptUrl = `${baseUrl}/dist/output.js`

      // Create and load the script
      const script = document.createElement('script')
      script.id = scriptId
      script.src = scriptUrl
      script.onerror = (e) => console.error('Failed to load custom-app:', e)
      document.head.appendChild(script)
    } else {
      console.error('custom-app load error: Could not find any asset link to infer base URL')
    }
  }, [])
  
  return null
}
