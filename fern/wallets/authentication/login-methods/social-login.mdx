# Social Login Authentication

Enable users to authenticate using their existing social media accounts. Social login provides a familiar and convenient authentication method that reduces friction and improves user conversion rates.

## Supported Providers

Account Kit supports the following social authentication providers:

* **Google** - Most popular social login option
* **Apple** - Required for iOS apps, growing on web
* **Facebook** - Wide user base, good conversion
* **Twitter/X** - Popular among certain demographics
* **Discord** - Great for gaming and community apps
* **GitHub** - Perfect for developer-focused applications

    class SocialAuthentication {
      constructor() {
        this.web3AuthSigner = new Web3AuthSigner({
          clientId: "YOUR_WEB3AUTH_CLIENT_ID",
          chainConfig: {
            chainNamespace: "eip155",
            chainId: "0xaa36a7", // Sepolia
          },
          web3AuthNetwork: "sapphire_mainnet", // or "testnet"
        });
      }

      async authenticateWithGoogle() {
        try {
          await this.web3AuthSigner.authenticate({
            loginProvider: "google",
            extraLoginOptions: {
              domain: "yourapp.com",
              login_hint: "user@example.com",
            },
          });
          
          return await this.createSmartAccount();
        } catch (error) {
          throw new Error(`Google authentication failed: ${error.message}`);
        }
      }

      async authenticateWithApple() {
        try {
          await this.web3AuthSigner.authenticate({
            loginProvider: "apple",
            extraLoginOptions: {
              domain: "yourapp.com",
            },
          });
          
          return await this.createSmartAccount();
        } catch (error) {
          throw new Error(`Apple authentication failed: ${error.message}`);
        }
      }

      async createSmartAccount() {
        const client = createAlchemySmartAccountClient({
          apiKey: "YOUR_ALCHEMY_API_KEY",
          chain: sepolia,
          signer: this.web3AuthSigner,
        });

        return client;
      }

      async getUserInfo() {
        return await this.web3AuthSigner.getUserInfo();
      }
    }

    // Usage
    const socialAuth = new SocialAuthentication();

    try {
      const client = await socialAuth.authenticateWithGoogle();

      console.log("User:", userInfo);

## Custom Social Login UI

### Custom Provider Buttons

```tsx
import { useAuthenticate } from "@alchemy/aa-alchemy/react";

const socialProviders = [
  {
    id: "google",
    name: "Google",
    icon: GoogleIcon,
    bgColor: "bg-white",
    textColor: "text-gray-900",
    borderColor: "border-gray-300",
  },
  {
    id: "apple", 
    name: "Apple",
    icon: AppleIcon,
    bgColor: "bg-black",
    textColor: "text-white",
    borderColor: "border-black",
  },
  {
    id: "facebook",
    name: "Facebook", 
    icon: FacebookIcon,
    bgColor: "bg-blue-600",
    textColor: "text-white",
    borderColor: "border-blue-600",
  },
  {
    id: "discord",
    name: "Discord",
    icon: DiscordIcon,
    bgColor: "bg-indigo-600",
    textColor: "text-white", 
    borderColor: "border-indigo-600",
  },
];

function CustomSocialLogin() {
  const { authenticate, isPending } = useAuthenticate();
  const [loadingProvider, setLoadingProvider] = useState<string | null>(null);

  const handleSocialLogin = async (providerId: string) => {
    setLoadingProvider(providerId);
    
    try {
      await authenticate({
        type: "oauth",
        authProviderId: providerId,
      });
    } catch (error) {
      console.error(`${providerId} login failed:`, error);
    } finally {
      setLoadingProvider(null);
    }
  };

  return (
    <div className="social-login-grid">
      {socialProviders.map((provider) => (
        <button
          key={provider.id}
          onClick={() => handleSocialLogin(provider.id)}
          disabled={isPending}
          className={`
            flex items-center justify-center gap-3 px-4 py-3 rounded-lg
            border transition-all duration-200 font-medium
            ${provider.bgColor} ${provider.textColor} ${provider.borderColor}
            hover:opacity-90 disabled:opacity-50 disabled:cursor-not-allowed
          `}
        >
          {loadingProvider === provider.id ? (
            <LoadingSpinner className="w-5 h-5" />
          ) : (
            <provider.icon className="w-5 h-5" />
          )}
          Continue with {provider.name}
        </button>
      ))}
    </div>
  );
}
```

## Error Handling

### Social Login Error States

```tsx
function SocialLoginWithErrorHandling() {
  const { authenticate, isPending } = useAuthenticate();
  const [error, setError] = useState<string | null>(null);

  const handleSocialLogin = async (provider: string) => {
    setError(null);
    
    try {
      await authenticate({
        type: "oauth",
        authProviderId: provider,
      });
    } catch (error: any) {
      // Handle specific social login errors
      switch (error.code) {
        case "POPUP_BLOCKED":
          setError("Popup was blocked. Please allow popups and try again.");
          break;
        case "USER_CANCELLED":
          setError("Login was cancelled.");
          break;
        case "NETWORK_ERROR":
          setError("Network error. Please check your connection.");
          break;
        case "PROVIDER_ERROR":
          setError(`${provider} authentication failed. Please try again.`);
          break;
        case "INSUFFICIENT_PERMISSIONS":
          setError("Required permissions were not granted.");
          break;
        default:
          setError("Authentication failed. Please try again.");
      }
    }
  };

  return (
    <div>
      {error && (
        <div className="error-banner">
          <AlertIcon />
          <span>{error}</span>
          <button onClick={() => setError(null)}>×</button>
        </div>
      )}
      
      <CustomSocialLogin onLogin={handleSocialLogin} />
    </div>
  );
}
```

### Profile Update Flow

```tsx
function ProfileUpdate() {
  const user = useUser();
  const [profile, setProfile] = useState({
    name: user?.name || "",
    email: user?.email || "",
  });

  const handleUpdateProfile = async () => {
    try {
      // Update profile via your backend API
      await fetch("/api/user/profile", {
        method: "PUT",
        headers: { "Content-Type": "application/json" },
        body: JSON.stringify(profile),
      });
      
      // Refresh user data
      window.location.reload();
    } catch (error) {
      console.error("Profile update failed:", error);
    }
  };

  return (
    <form onSubmit={handleUpdateProfile}>
      <input
        type="text"
        value={profile.name}
        onChange={(e) => setProfile(prev => ({ ...prev, name: e.target.value }))}
        placeholder="Full Name"
      />
      
      <input
        type="email"
        value={profile.email}
        onChange={(e) => setProfile(prev => ({ ...prev, email: e.target.value }))}
        placeholder="Email Address"
      />
      
      <button type="submit">Update Profile</button>
    </form>
  );
}
```
