# Email Magic Link Authentication

Enable passwordless authentication using magic links sent to users' email addresses. This provides a secure and user-friendly way to authenticate without requiring password management.

## Overview

Magic link authentication sends a unique, time-limited link to the user's email address. When clicked, the link automatically authenticates the user and creates or accesses their smart wallet.

**Benefits:**

* **Passwordless** - No need to remember passwords
* **Secure** - Links are time-limited and single-use
* **User-friendly** - Familiar email-based flow
* **Cross-device** - Works on any device with email access

## Handling Magic Link Callbacks

### React Router Setup

```tsx
import { useEffect } from "react";
import { useSearchParams, useNavigate } from "react-router-dom";
import { useAuthenticate } from "@alchemy/aa-alchemy/react";

function AuthCallback() {
  const [searchParams] = useSearchParams();
  const navigate = useNavigate();
  const { completeAuthentication } = useAuthenticate();
  
  useEffect(() => {
    const handleCallback = async () => {
      const token = searchParams.get("magic_credential");
      
      if (token) {
        try {
          await completeAuthentication({ token });
          navigate("/dashboard");
        } catch (error) {
          console.error("Authentication failed:", error);
          navigate("/auth?error=auth_failed");
        }
      }
    };
    
    handleCallback();
  }, [searchParams, navigate, completeAuthentication]);
  
  return (
    <div className="auth-callback">
      <div className="spinner" />
      <p>Completing authentication...</p>
    </div>
  );
}
```

### Next.js API Route

```typescript
// pages/api/auth/callback.ts
import type { NextApiRequest, NextApiResponse } from "next";
import { MagicSigner } from "@alchemy/aa-signers/magic";

export default async function handler(
  req: NextApiRequest,
  res: NextApiResponse
) {
  if (req.method !== "POST") {
    return res.status(405).json({ message: "Method not allowed" });
  }

  try {
    const { token } = req.body;
    
    const magicSigner = new MagicSigner({
      apiKey: process.env.MAGIC_API_KEY!,
    });
    
    // Validate the magic link token
    const user = await magicSigner.validateToken(token);
    
    if (user) {
      // Set session or JWT token
      res.setHeader("Set-Cookie", `auth-token=${token}; Path=/; HttpOnly`);
      return res.status(200).json({ success: true, user });
    }
    
    return res.status(401).json({ message: "Invalid token" });
  } catch (error) {
    console.error("Auth callback error:", error);
    return res.status(500).json({ message: "Authentication failed" });
  }
}
```

## Error Handling

### Common Error Scenarios

```tsx
function MagicLinkAuthWithErrorHandling() {
  const { authenticate, isPending } = useAuthenticate();
  const [error, setError] = useState<string | null>(null);

  const handleAuthentication = async (email: string) => {
    setError(null);
    
    try {
      await authenticate({
        type: "email",
        email,
      });
    } catch (error: any) {
      // Handle specific error types
      switch (error.code) {
        case "INVALID_EMAIL":
          setError("Please enter a valid email address");
          break;
        case "RATE_LIMITED":
          setError("Too many attempts. Please try again later");
          break;
        case "DOMAIN_NOT_ALLOWED":
          setError("Email domain not supported");
          break;
        case "NETWORK_ERROR":
          setError("Network error. Please check your connection");
          break;
        default:
          setError("Authentication failed. Please try again");
      }
    }
  };

  return (
    <div>
      {error && (
        <div className="error-message">
          {error}
        </div>
      )}
      {/* Rest of component */}
    </div>
  );
}
```
