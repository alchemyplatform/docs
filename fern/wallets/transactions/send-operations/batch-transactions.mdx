# Batch Transactions

Batch transactions allow you to execute multiple operations atomically in a single user operation. This reduces gas costs, improves efficiency, and ensures all operations succeed or fail together.

## Overview

Batch transactions enable you to:

* **Reduce gas costs** - Single user operation for multiple actions
* **Atomic execution** - All operations succeed or fail together
* **Improved UX** - Single signature for multiple transactions
* **Efficient bundling** - Optimize blockchain interactions

## Common Batch Transaction Patterns

### DeFi Operations

```tsx
async function defiBatchTransaction(client: SmartAccountClient) {
  const operations = [
    // 1. Approve USDC spending
    {
      target: USDC_ADDRESS,
      data: encodeFunctionData({
        abi: erc20Abi,
        functionName: "approve",
        args: [UNISWAP_ROUTER_ADDRESS, parseUnits("1000", 6)],
      }),
      value: 0n,
    },
    
    // 2. Swap USDC for ETH
    {
      target: UNISWAP_ROUTER_ADDRESS,
      data: encodeFunctionData({
        abi: uniswapRouterAbi,
        functionName: "swapExactTokensForETH",
        args: [
          parseUnits("1000", 6), // amount in
          0n, // min amount out
          [USDC_ADDRESS, WETH_ADDRESS], // path
          client.account.address, // to
          Math.floor(Date.now() / 1000) + 300, // deadline
        ],
      }),
      value: 0n,
    },
    
    // 3. Deposit ETH to lending protocol
    {
      target: AAVE_POOL_ADDRESS,
      data: encodeFunctionData({
        abi: aavePoolAbi,
        functionName: "deposit",
        args: [
          ETH_ADDRESS,
          parseEther("0.5"),
          client.account.address,
          0,
        ],
      }),
      value: parseEther("0.5"),
    },
  ];

  return await client.sendUserOperation({ uo: operations });
}
```

## Advanced Batch Patterns

### Conditional Batch Execution

```tsx
async function conditionalBatch(client: SmartAccountClient) {
  // Check conditions before batching
  const balance = await client.readContract({
    address: TOKEN_ADDRESS,
    abi: erc20Abi,
    functionName: "balanceOf",
    args: [client.account.address],
  });

  const operations = [];

  // Only add operations based on conditions
  if (balance > parseEther("100")) {
    operations.push({
      target: TOKEN_ADDRESS,
      data: encodeFunctionData({
        abi: erc20Abi,
        functionName: "transfer",
        args: ["0xRecipient", parseEther("50")],
      }),
      value: 0n,
    });
  }

  if (operations.length > 0) {
    return await client.sendUserOperation({ uo: operations });
  }
}
```

### Dynamic Batch Building

```tsx
class BatchBuilder {
  private operations: Array<{
    target: string;
    data: string;
    value: bigint;
  }> = [];

  addEthTransfer(to: string, amount: bigint) {
    this.operations.push({
      target: to,
      data: "0x",
      value: amount,
    });
    return this;
  }

  addTokenTransfer(tokenAddress: string, to: string, amount: bigint) {
    this.operations.push({
      target: tokenAddress,
      data: encodeFunctionData({
        abi: erc20Abi,
        functionName: "transfer",
        args: [to, amount],
      }),
      value: 0n,
    });
    return this;
  }

  addContractCall(
    contractAddress: string,
    abi: any,
    functionName: string,
    args: any[],
    value: bigint = 0n
  ) {
    this.operations.push({
      target: contractAddress,
      data: encodeFunctionData({
        abi,
        functionName,
        args,
      }),
      value,
    });
    return this;
  }

  async execute(client: SmartAccountClient) {
    if (this.operations.length === 0) {
      throw new Error("No operations to execute");
    }

    const userOpHash = await client.sendUserOperation({
      uo: this.operations,
    });

    // Clear operations after execution
    this.operations = [];

    return userOpHash;
  }
}

// Usage
const batch = new BatchBuilder()
  .addEthTransfer("0x1111...", parseEther("0.1"))
  .addTokenTransfer(TOKEN_ADDRESS, "0x2222...", parseEther("100"))
  .addContractCall(
    CONTRACT_ADDRESS,
    contractAbi,
    "someFunction",
    ["arg1", "arg2"]
  );

await batch.execute(client);
```

## Error Handling and Recovery

### Batch Transaction Validation

```tsx
async function validateBatchOperations(
  client: SmartAccountClient,
  operations: Array<{ target: string; data: string; value: bigint }>
) {
  // Validate each operation before sending
  for (const [index, operation] of operations.entries()) {
    try {
      // Simulate the operation
      await client.simulateUserOperation({
        uo: [operation],
      });
    } catch (error) {
      throw new Error(`Operation ${index} validation failed: ${error.message}`);
    }
  }

  // If all validations pass, send the batch
  return await client.sendUserOperation({ uo: operations });
}
```

### Partial Failure Handling

```tsx
async function batchWithFallback(
  client: SmartAccountClient,
  operations: Array<{ target: string; data: string; value: bigint }>
) {
  try {
    // Try to send all operations in batch
    return await client.sendUserOperation({ uo: operations });
  } catch (error) {
    console.warn("Batch failed, trying individual operations:", error);

    // Fall back to individual operations
    const results = [];
    for (const operation of operations) {
      try {
        const result = await client.sendUserOperation({ uo: [operation] });
        results.push({ success: true, result });
      } catch (opError) {
        results.push({ success: false, error: opError });
      }
    }

    return results;
  }
}
```

## Gas Optimization

### Estimating Batch Gas

```tsx
async function estimateBatchGas(
  client: SmartAccountClient,
  operations: Array<{ target: string; data: string; value: bigint }>
) {
  try {
    // Estimate gas for the batch
    const gasEstimate = await client.estimateUserOperationGas({
      uo: operations,
    });

    console.log("Batch gas estimate:", {
      callGasLimit: gasEstimate.callGasLimit,
      verificationGasLimit: gasEstimate.verificationGasLimit,
      preVerificationGas: gasEstimate.preVerificationGas,
    });

    // Calculate total gas cost
    const totalGas = 
      gasEstimate.callGasLimit +
      gasEstimate.verificationGasLimit +
      gasEstimate.preVerificationGas;

    return {
      totalGas,
      gasPrice: gasEstimate.maxFeePerGas,
      estimatedCost: totalGas * gasEstimate.maxFeePerGas,
    };
  } catch (error) {
    console.error("Gas estimation failed:", error);
    throw error;
  }
}
```

### Optimizing Operation Order

```tsx
function optimizeBatchOrder(operations: BatchOperation[]) {
  // Sort operations for optimal gas usage
  return operations.sort((a, b) => {
    // ETH transfers first (lower gas)
    if (a.data === "0x" && b.data !== "0x") return -1;
    if (a.data !== "0x" && b.data === "0x") return 1;
    
    // Sort by gas complexity
    return a.data.length - b.data.length;
  });
}
```

## Best Practices

### Performance Tips

1. **Order operations efficiently** - Simple operations first
2. **Validate before sending** - Use simulation to catch errors
3. **Optimize gas limits** - Set appropriate limits for each operation
4. **Handle failures gracefully** - Implement fallback strategies

### Security Considerations

1. **Validate all inputs** - Ensure addresses and amounts are correct
2. **Check balances** - Verify sufficient funds before operations
3. **Use timeouts** - Set appropriate deadlines for time-sensitive operations
4. **Monitor for MEV** - Be aware of potential front-running
