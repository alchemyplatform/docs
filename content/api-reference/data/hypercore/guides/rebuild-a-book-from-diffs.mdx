# Rebuild a book from diffs

`l2BookDiff` sends a full snapshot when you subscribe, then only the price levels that change. This guide keeps a local book correct and recovers when something goes wrong.

```js
// You provide: backoffWithJitter, onFatal.
const RETRYABLE = new Set(["book_not_ready"]);   // temporary; retry after a short delay
const books = {};      // coin -> { bids: Map<px, level>, asks: Map<px, level> }
let subId = null;      // current subscription; messages for any other ID are ignored
let generation = 0;
let lastSeq = 0;
let ready = false;     // true once the current subscription's snapshot has arrived
let halted = false;    // set when an error needs a fix, not a retry

function connect() {
  const ws = new WebSocket("wss://hypercore-mainnet.g.alchemy.com/v2/<API_KEY>");
  let closed = false;

  // Start a fresh subscription. A new ID means late messages from the old one are ignored.
  function subscribe() {
    if (closed) return;
    if (subId) ws.send(JSON.stringify({ method: "unsubscribe", subscriptionId: subId }));
    subId = `book-${++generation}`;
    lastSeq = 0;
    resetBooks();
    ws.send(JSON.stringify({
      method: "subscribe",
      subscriptionId: subId,
      subscription: { type: "l2BookDiff", coins: ["BTC", "ETH"], nLevels: 20 },
    }));
  }

  function halt(error) {                       // e.g. invalid_params or limit_reached
    halted = true;
    onFatal(error);
    ws.close();
  }

  ws.onmessage = ({ data }) => {
    let msg;
    try {
      msg = JSON.parse(data);
    } catch {
      return ws.close();                       // malformed message: reconnect
    }
    if (closed) return;
    if (msg.channel === "connected") {
      subId = null;                            // new connection, nothing to unsubscribe
      return subscribe();
    }

    // Errors come as a subscription error (with our ID) or a connection-level error (no ID).
    if (msg.error && (msg.channel === "error" || msg.subscriptionId === subId)) {
      if (msg.error.code === "book_reset") return resetBooks();  // a fresh snapshot follows
      if (RETRYABLE.has(msg.error.code)) {
        subId = null;                          // that subscription was never created
        return setTimeout(subscribe, 1000);
      }
      return halt(msg.error);
    }
    if (msg.subscriptionId !== subId) return;  // stale or unrelated subscription
    if (msg.seq === undefined) return;         // acknowledgement

    if (msg.seq !== lastSeq + 1) return subscribe();  // missed a message: start over
    lastSeq = msg.seq;

    if (msg.data.isSnapshot) ready = true;
    if (!ready) return;                        // never apply diffs without a snapshot

    for (const { coin, levels } of msg.data.books) {
      if (msg.data.isSnapshot || !books[coin]) books[coin] = { bids: new Map(), asks: new Map() };
      const [bids, asks] = levels;
      apply(books[coin].bids, bids);
      apply(books[coin].asks, asks);
    }
  };

  ws.onclose = () => {
    closed = true;
    if (!halted) setTimeout(connect, backoffWithJitter());
  };
}

function resetBooks() {
  for (const coin in books) delete books[coin];
  ready = false;
}

function apply(side, levels) {
  for (const l of levels) {
    if (Number(l.sz) === 0 || l.n === 0) side.delete(l.px);
    else side.set(l.px, l);
  }
}

connect();
```

## Rules

* Build from this stream's own snapshot, not from a separate `l2Book` subscription.
* Keep `nSigFigs`, `mantissa`, and `nLevels` the same for the life of the book.
* Sort when you read: bids high to low, asks low to high. Keep the top `nLevels` per side.
* Detect missed messages with `seq`, not `blockNumber`. Quiet markets skip blocks.
* When you start over, use a new `subscriptionId`. Messages already in flight for the old subscription then can't be mistaken for the new one.
* Retry `book_not_ready` after a short delay. Other errors, such as `invalid_params` or `limit_reached`, need a change to the request or your limits, so the example stops and reports them through `onFatal`. See [errors and limits](/docs/chains/websockets/hypercore/errors-and-limits).

The same approach works for `l4BookUpdates`, keyed by order ID instead of price. See [l4BookUpdates](/docs/chains/websockets/hypercore/streams/l4-book-updates).
